document.getElementById("checkout-form").addEventListener("submit", async function(event) { event.preventDefault(); const cart = readCart(); const status = document.getElementById("checkout-status"); const submitButton = document.getElementById("place-order"); if (!cart.length) { status.textContent = "Your cart is empty. Add a product first."; return; } const form = event.currentTarget; if (!form.reportValidity()) return; const paymentMethod = form.elements.payment_method.value; const payload = { customer_name: form.elements.customer_name.value.trim(), mobile: form.elements.mobile.value.trim(), address: form.elements.address.value.trim(), city: form.elements.city.value.trim(), state: form.elements.state.value.trim(), pincode: form.elements.pincode.value.trim(), items: cart.map(function(item) { return { product_id: Number(item.id), quantity: Number(item.quantity) }; }) }; if (payload.items.some(function(item) { return !Number.isInteger(item.product_id) || item.product_id < 1 || !Number.isInteger(item.quantity) || item.quantity < 1; })) { status.textContent = "Cart mein product details sahi nahi hain. Please refresh karke dobara try karein."; return; } function showOrderSuccess(orderNumber, total, paymentText) { localStorage.removeItem("ashhaz_cart"); updateCartDisplay(); form.reset(); form.hidden = true; status.textContent = ""; const success = document.createElement("p"); success.className = "cart-empty"; success.textContent = "Order successfully placed! Order Number: " + orderNumber + ". Total: " + money(total) + ". Payment: " + paymentText; cartItemsBox.replaceChildren(success); document.getElementById("cart-total").textContent = money(0); document.getElementById("show-checkout").hidden = true; } let paymentWindowOpened = false; submitButton.disabled = true; submitButton.textContent = "Please wait..."; status.textContent = "Your order is being processed. Please wait."; try { // CASH ON DELIVERY if (paymentMethod === "COD") { submitButton.textContent = "Submitting COD order..."; const response = await fetch("/api/orders", { method: "POST", headers: {"Content-Type": "application/json"}, body: JSON.stringify(payload) }); const result = await response.json().catch(function() { return {}; }); if (!response.ok || !result.ok) { throw new Error(result.error || "Order save nahi ho saka. Please try again."); } showOrderSuccess( result.order_number, result.total_amount, "Cash on Delivery" ); return; } // RAZORPAY ONLINE PAYMENT if (paymentMethod === "Razorpay") { if (typeof window.Razorpay !== "function") { throw new Error("Payment service load nahi hui. Page refresh karke dobara try karein."); } submitButton.textContent = "Creating payment..."; const response = await fetch("/api/create-payment-order", { method: "POST", headers: {"Content-Type": "application/json"}, body: JSON.stringify(payload) }); const paymentOrder = await response.json().catch(function() { return {}; }); if (!response.ok || !paymentOrder.ok) { throw new Error(paymentOrder.error || "Payment order create nahi ho saka."); } const options = { key: paymentOrder.key_id, amount: paymentOrder.amount, currency: paymentOrder.currency, name: "Ashhaz International Inc", description: "Online Shopping Order", order_id: paymentOrder.order_id, prefill: { name: payload.customer_name, contact: payload.mobile }, handler: async function(paymentResponse) { submitButton.disabled = true; submitButton.textContent = "Verifying payment..."; status.textContent = "Payment received. Verification ho rahi hai. Please wait."; try { const verifyResponse = await fetch("/api/verify-payment", { method: "POST", headers: {"Content-Type": "application/json"}, body: JSON.stringify({ razorpay_order_id: paymentResponse.razorpay_order_id, razorpay_payment_id: paymentResponse.razorpay_payment_id, razorpay_signature: paymentResponse.razorpay_signature }) }); const verification = await verifyResponse.json().catch(function() { return {}; }); if (!verifyResponse.ok || !verification.ok) { throw new Error( verification.error || "Payment verification nahi ho saki. Support se contact karein." ); } showOrderSuccess( paymentOrder.order_number, Number(paymentOrder.amount) / 100, "Paid Online" ); } catch (error) { status.textContent = error.message || "Payment verification mein problem hui. Please contact support."; submitButton.disabled = false; submitButton.textContent = "Continue to Payment"; } }, modal: { ondismiss: function() { status.textContent = "Payment window band ho gayi. Aap dobara payment try kar sakte hain."; submitButton.disabled = false; submitButton.textContent = "Continue to Payment"; } }, theme: { color: "#c3a064" } }; const razorpay = new window.Razorpay(options); razorpay.on("payment.failed", function(paymentError) { status.textContent = paymentError.error?.description || "Payment failed. Please try again."; submitButton.disabled = false; submitButton.textContent = "Continue to Payment"; }); razorpay.open(); paymentWindowOpened = true; status.textContent = "Razorpay window mein payment complete karein."; return; } throw new Error("Please select a valid payment method."); } catch (error) { status.textContent = error.message || "Order process nahi ho saka. Please try again."; } finally { if (!paymentWindowOpened) { submitButton.disabled = false; submitButton.textContent = "Continue to Payment"; } } });
